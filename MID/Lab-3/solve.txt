###2
.model small
.stack 100h
.data
msgHello db 0ah,0dh, "Hello, World$"
counter  db 0          ; counter to track number of prints

.code
main proc
    mov ax, @data
    mov ds, ax           ; initialize data segment

start:
    ; print "Hello, World"
    mov ah, 09h
    lea dx, msgHello
    int 21h

    ; increment counter
    mov al, counter
    inc al
    mov counter, al

    ; compare counter with 5
    cmp counter, 5
    je done              ; if counter == 5, exit loop
    jmp start            ; else, repeat

done:
    mov ah, 4Ch          ; exit program
    int 21h
main endp
end main

.model small
.stack 100h
.data
prompt db 'Enter a number (0-9): $'
lessMsg db 10,13,'Less than 5$'
equalMsg db 10,13,'Equal to 5$'
greaterMsg db 10,13,'Greater than 5$'
invalidMsg db 10,13,'Invalid input$'

.code
main proc
    mov ax, @data
    mov ds, ax

    mov cl, 5           ; CL contains 5

    ; print prompt
    mov ah, 9
    lea dx, prompt
    int 21h

    ; read character from user
    mov ah, 1
    int 21h
    sub al, '0'          ; convert ASCII to number
    mov bl, al           ; store input in BL

    ; check if input is valid (0-9)
    cmp bl, 0
    jl invalid
    cmp bl, 9
    jg invalid

    ; compare input with CL
    cmp bl, cl
    je equal
    jl less
    jg greater

less:
    mov ah, 9
    lea dx, lessMsg
    int 21h
    jmp exit

equal:
    mov ah, 9
    lea dx, equalMsg
    int 21h
    jmp exit

greater:
    mov ah, 9
    lea dx, greaterMsg
    int 21h
    jmp exit

invalid:
    mov ah, 9
    lea dx, invalidMsg
    int 21h

exit:
    mov ah, 4Ch
    int 21h

main endp
end main

.model small
.stack 100h

.data
prompt      db 'Enter password: $'
correctPass db 'mypassword', 0
successMsg  db 10,13,'Correct Password$'
failMsg     db 10,13,'Wrong Password$'
inputBuff   db 20 dup(?)

.code
main proc
    mov ax, @data
    mov ds, ax

    mov ah, 9
    lea dx, prompt
    int 21h

    mov si, offset inputBuff

read_loop:
    mov ah, 1
    int 21h
    cmp al, 13
    je compare_password
    mov [si], al
    inc si
    jmp read_loop

compare_password:
    mov byte ptr [si], 0

    mov si, offset inputBuff
    mov di, offset correctPass

compare_loop:
    mov al, [si]
    mov bl, [di]
    cmp al, bl
    jne wrong
    cmp al, 0
    je correct
    inc si
    inc di
    jmp compare_loop

correct:
    mov ah, 9
    lea dx, successMsg
    int 21h
    jmp exit

wrong:
    mov ah, 9
    lea dx, failMsg
    int 21h

exit:
    mov ah, 4Ch
    int 21h

main endp
end main
