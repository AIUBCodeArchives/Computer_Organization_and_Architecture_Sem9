.MODEL SMALL
.STACK 100H
.DATA 
 
C DB ?
D DB ?

.CODE
MAIN PROC
    ;load data
    MOV AX,@DATA
    MOV DS,AX 
    ;take input for C
    MOV AH,1
    INT 21H
    MOV C,AL
    ;take input for D
    MOV AH,1
    INT 21H
    MOV D,AL
    
    
    jmp  Exit  
    
    ;Exchange C and D
    MOV BH,C
    XCHG BH,D
    MOV C,BH
   
   ;print space for input C to newline
    MOV AH,2
    MOV DL,10
    INT 21H
    MOV DL,13
    INT 21H
    ;print space for input D to newline
    MOV AH,2
    MOV DL,10
    INT 21H
    MOV DL,13
    INT 21H
    ;print C
    MOV AH,2
    MOV DL,C
    INT 21H
    ;print D
    MOV AH,2
    MOV DL,D
    INT 21H    
    
    Exit:
    ;terminate program
    MOV AH,4CH
    INT 21H
    
    MAIN ENDP
END MAIN
